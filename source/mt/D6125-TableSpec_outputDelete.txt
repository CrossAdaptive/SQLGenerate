void
_outputDelete( TableSpec* self, IO* out )
{
    String* primary = null;

    const Array* entries = Dictionary_getEntries( self->members );
          int    n       = Array_getCount( entries );

    for ( int i=0; i < n; i++ )
    {
        const Entry*  entry = (const Entry*) Array_get_index( entries, i );
        const String* name  =                Entry_getKey   ( entry      );
        const Type*   type  = (const Type*)  Entry_getValue ( entry      );

        if ( type->primary )
        {
            primary = String_new( String_getContent( name ) );
        }
    }

    if ( primary )
    {
        const char* _primary = String_getContent( primary );

        IO_print( out, "DELIMITER //"                  );
        IO_print( out, "\n" );

        IO_print( out, "CREATE PROCEDURE "             );
        IO_print( out, "spgen_"                        );
        IO_print( out, String_getChars( self->name )   );
        IO_print( out, "_delete"                       );
        IO_print( out, "\n"                            );
        IO_print( out, "("                             );

        _outputDeleteParameters( self, out );

        IO_print( out, "\n"                            );

        IO_print( out, "\n"                            );

        IO_print( out, "BEGIN"                         );

        IO_print( out, "\n"                            );
        IO_print( out, "END"                           );
        IO_print( out, "\n"                            );
        IO_print( out, "//"                            );
        IO_print( out, "\n"                            );
        IO_print( out, "DELIMITER ;"                   );
        IO_print( out, "\n"                            );

    Delete( &primary );
}

~source/c/TableSpec.c~
void
_outputDeleteParameters( TableSpec* self, IO* out )
{
    const Array* entries = Dictionary_getEntries( self->members );
          int    max     = 0;
          int    n       = Array_getCount( entries );

    for ( int i=0; i < n; i++ )
    {
        const Entry*  entry = (const Entry*) Array_get_index( entries, i );
        const String* name  =                Entry_getKey( entry );
        const Type*   type  = (const Type*)  Entry_getValue ( entry      );
              int     len   =                String_getLength( name );

        if ( type->foreign || type->primary )
        {
            max = (len > max) ? len : max;
        }
    }

    const char* sep = "\n";

    for ( int i=0; i < n; i++ )
    {
        const Entry*  entry = (const Entry*)  Array_get_index( entries, i );
        const String* name  =                 Entry_getKey   ( entry      );
        const Type*   type  = (const Type*)   Entry_getValue ( entry      );

        if ( type->foreign || type->primary )
        {
            IO_print( out, sep                            );

            String* _name = String_new_length( String_getContent( name ), max );
            {
                IO_print( out, "\t"                       );
                IO_print( out, "$"                        );
                IO_print( out, String_getContent( _name ) );
                IO_print( out, " "                        );
            }
            Delete( &_name );

            if ( type && type->translated )
            {
                IO_print( out, String_getContent( type->translated ) );
            }

            sep = ",\n";
        }
    }
    IO_print( out, "\n" );
}

~source/c/TableSpec.c~
void
_outputDeleteUpdateStatement( TableSpec* self, IO* out )
{
    const String* prefix = (const String*) Dictionary_find_key( self->options, "Prefix" );

    if ( prefix )
    {
        const Array* entries = Dictionary_getEntries( self->members           );
              int    max     = String_getLength( prefix ) + 7;
              int    n       = Array_getCount( entries );

        for ( int i=0; i < n; i++ )
        {
            const Entry*  entry = (const Entry*) Array_get_index( entries, i );
            const String* name  = Entry_getKey( entry );
            const Type*   type  = (const Type*)   Entry_getValue ( entry      );
                  int     len   = String_getLength( name );

            if ( type->foreign || type->primary )
            {
                max = (len > max) ? len : max;
            }
        }

        IO_print( out, "\t"                             );
        IO_print( out, "UPDATE "                        );
        IO_print( out, String_getContent( self->name ) );
        IO_print( out, "\n"                             );

        IO_print( out, "\t"                             );
        IO_print( out, "SET    "                        );
        IO_print( out, String_getContent( prefix )      );
        IO_print( out, "deleted"                        );
        IO_print( out, " = NOW()"                       );
        IO_print( out, "\n"                             );

        IO_print( out, "\t"                             );
        IO_print( out, "WHERE  "                        );
        IO_print( out, String_getContent( prefix )      );
        IO_print( out, "deleted"                        );
        IO_print( out, " = 0"                           );
        IO_print( out, "\n"                             );

        for ( int i=0; i < n; i++ )
        {
            const Entry*  entry = (const Entry*)  Array_get_index( entries, i );
            const String* name  = (const String*) Entry_getKey   ( entry      );
            const Type*   type  = (const Type*)   Entry_getValue ( entry      );

            if ( type->foreign || type->primary )
            {
                String* _name = String_new_length( String_getContent( name ), max );
                {
                    IO_print( out, "\t"                       );
                    IO_print( out, "AND    "                  );
                    IO_print( out, String_getContent( _name ) );
                    IO_print( out, " = "                      );
                    IO_print( out, "$"                        );
                    IO_print( out, String_getContent( _name ) );
                    IO_print( out, "\n"                       );
                }
            }
        }

        IO_print( out, "\t" );
        IO_print( out, ";"  );
        IO_print( out, "\n" );
    }
}
~
