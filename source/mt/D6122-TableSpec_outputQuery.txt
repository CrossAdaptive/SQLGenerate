void
_outputQuery( TableSpec* self, IO* out )
{
    switch( self->output )
    {
    case PSQL:
        break;

    default:
        IO_print( out, "DELIMITER //"              );
        IO_print( out, "\n" );
    }

    IO_print( out, "CREATE PROCEDURE "             );
    IO_print( out, "spgen_"                        );
    IO_print( out, String_getChars( self->name )   );
    IO_print( out, "\n"                            );
    IO_print( out, "("                             );
    IO_print( out, "\n"                            );

    _outputQueryParameters( self, out );

    IO_print( out, ")"                             );
    IO_print( out, "\n"                            );

    switch( self->output )
    {
    case PSQL:
        IO_print( out, "LANGUAGE plpgsql" );
        IO_print( out, "\n"               );
        IO_print( out, "AS $$"            );
        IO_print( out, "\n"               );
        break;

    default:
        IO_print( out, "SQL SECURITY INVOKER"          );
        IO_print( out, "\n"                            );
        IO_print( out, "COMMENT 'Generated by sqlgen'" );
        IO_print( out, "\n"                            );
        ;
    }

    IO_print( out, "BEGIN"                         );
    IO_print( out, "\n"                            );
    IO_print( out, "\n"                            );

    switch( self->output )
    {
    case PSQL:
        IO_print( out, "\t"                                               );
        IO_print( out, "IF p_limit = 0 THEN p_limit := 2147483647; END IF;" );
        IO_print( out, "\n"                                               );
        break;

    default:
        IO_print( out, "\t"                                                  );
        IO_print( out, "IF 0 = $limit THEN SET $limit = 2147483647; END IF;" );
        IO_print( out, "\n"                                                  );
    }
    IO_print( out, "\n"                            );

    _outputQueryIfStatement( self, out );

    IO_print( out, "END"                           );
    IO_print( out, "\n"                            );

    switch( self->output )
    {
    case PSQL:
        IO_print( out, "$$;"                       );
        IO_print( out, "\n"                        );
        break;

    default:
        IO_print( out, "//"                        );
        IO_print( out, "\n"                        );
        IO_print( out, "DELIMITER ;"               );
        IO_print( out, "\n" );
    }
    IO_flushAll();
}

void
_outputQueryParameters( TableSpec* self, IO* out )
{
    const String* filter  = (const String*) Dictionary_find_key  ( self->options, "Filter" );
    const char*   prefix  = (PSQL == self->output) ? "IN p_" : "$";

    Array* parts = String_toArray_separator( filter, ';' );
    int    max   = CharString_length( "offset" ); // size of largest default parameter.
    int    n     = Array_getCount( parts );

    for ( int i=0; i < n; i++ )
    {
        const String* part = (const String*) Array_get_index( parts, i );
              int     len  = String_getLength( part );

        max = (max > len) ? max : len;
    }

    for ( int i=0; i < n; i++ )
    {
        const String* part = (const String*) Array_get_index( parts, i );
        const Type*   type = (const Type*)   Dictionary_find_key( self->members, String_getContent( part ) );
        const char*  _type = "TEXT";

        if ( type )
        {
            if ( String_contentEquals( type->translated, "SERIAL" ) )
            {
                _type = "INT";
            }
            else
            {
                _type = String_getContent( type->translated );
            }
        }

        String* _part = String_new_length( String_getContent( part ), max );

        IO_print( out, "\t"                                  );
        IO_print( out, prefix                                );
        IO_print( out, String_getContent( _part            ) );
        IO_print( out, " "                                   );
        IO_print( out, _type                                 );
        IO_print( out, ","                                   );
        IO_print( out, "\n"                                  );

        Delete( &_part );
    }

    String* order  = String_new_length( "order",  max );
    String* limit  = String_new_length( "limit",  max );
    String* offset = String_new_length( "offset", max );

    IO_print( out, "\t"                        );
    IO_print( out, prefix                      );
    IO_print( out, String_getContent( order )  );
    IO_print( out, " "                         );
    IO_print( out, "TEXT"                      );
    IO_print( out, ","                         );
    IO_print( out, "\n"                        );

    IO_print( out, "\t"                        );
    IO_print( out, prefix                      );
    IO_print( out, String_getContent( limit )  );
    IO_print( out, " "                         );
    IO_print( out, "INT"                       );
    IO_print( out, ","                         );
    IO_print( out, "\n"                        );

    IO_print( out, "\t"                        );
    IO_print( out, prefix                      );
    IO_print( out, String_getContent( offset ) );
    IO_print( out, " "                         );
    IO_print( out, "INT"                       );
    IO_print( out, "\n"                        );
}

void
_outputQueryIfStatement( TableSpec* self, IO* out )
{
    const String* order = (const String*) Dictionary_find_key( self->options, "OrderBy" );
    const String* dflt  = null;
          int     i     = 0;

    if ( order )
    {
        Array* options = String_toArray_separator( order, ';' );
        int    n       = Array_getCount( options );

        if ( 0 < n )
        {
            dflt = (String*) Array_get_index( options, 0 );
        }

        for ( i=0; i < n; i++ )
        {
            const String* option = (String*) Array_get_index( options, i );

            _outputQueryIfStatementFor( self, out, i, String_getContent( option ), String_getContent( dflt ) );
        }
    }

    if ( dflt )
    {
        _outputQueryIfStatementFor( self, out, i, "", String_getContent( dflt ) );
    }
    else
    {
        _outputQueryIfStatementFor( self, out, i, "", "" );
    }

    IO_print( out, "\t"                                                               );
    IO_print( out, "ELSE"                                                             );
    IO_print( out, "\n"                                                               );
    IO_print( out, "\n"                                                               );
    IO_print( out, "\t"                                                               );

    switch( self->output )
    {
    case PSQL:
        IO_print( out, "\t"                                                               );
        IO_print( out, "RAISE EXCEPTION 'INVALID ORDER BY IN "                            );
        IO_print( out, "spgen_"                                                           );
        IO_print( out, String_getContent( self->name )                                    );
        IO_print( out, "';"                                                               );
        IO_print( out, "\n"                                                               );
        break;

    default:
        IO_print( out, "\t"                                                               );
        IO_print( out, "SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'INVALID ORDER BY IN " );
        IO_print( out, "spgen_"                                                           );
        IO_print( out, String_getContent( self->name )                                    );
        IO_print( out, "';"                                                               );
        IO_print( out, "\n"                                                               );
        break;
    }

    IO_print( out, "\n"                                                               );
    IO_print( out, "\t"                                                               );
    IO_print( out, "END IF;"                                                          );
    IO_print( out, "\n"                                                               );
    IO_print( out, "\n"                                                               );
}

void
_outputQueryIfStatementFor( TableSpec* self, IO* out, int i, const char* order_by, const char* dflt )
{
    const char*   dollar = (PSQL == self->output) ? "p_" : "$";
    const String* view   = (const String*) Dictionary_find_key( self->options, "View" );

    IO_print( out, "\t" );

    if ( 0 == i )
    {
        IO_print( out, "IF " );
    }
    else
    {
        IO_print( out, "ELSEIF " );
    }

    IO_print( out, "'"                 );
    IO_print( out, order_by            );
    IO_print( out, "'"                 );

    IO_print( out, " = "               );
    IO_print( out, dollar              );
    IO_print( out, "order THEN"        );
    IO_print( out, "\n"                );

    IO_print( out, "\t"                );
    IO_print( out, "\t"                );
    IO_print( out, "SELECT   *"        );
    IO_print( out, "\n"                );

    IO_print( out, "\t"                );
    IO_print( out, "\t"                );
    IO_print( out, "FROM     "         );

    if ( view )
    {
        IO_print( out, String_getContent( view ) );
    }
    else
    {
        IO_print( out, String_getContent( self->name ) );
    }
    IO_print( out, "\n"                );

    _outputQueryIfStatementWhereClauses( self, out );

    if ( CharString_length( order_by ) )
    {
        IO_print( out, "\t"            );
        IO_print( out, "\t"            );
        IO_print( out, "ORDER BY "     );
        IO_print( out, order_by        );
        IO_print( out, "\n"            );
    }
    else
    if ( CharString_length( dflt ) )
    {
        IO_print( out, "\t"            );
        IO_print( out, "\t"            );
        IO_print( out, "ORDER BY "     );
        IO_print( out, dflt            );
        IO_print( out, "\n"            );
    }

    IO_print( out, "\t"                );
    IO_print( out, "\t"                );
    IO_print( out, "LIMIT   "          );
    IO_print( out, dollar              );
    IO_print( out, "limit"             );
    IO_print( out, "\n"                );

    IO_print( out, "\t"                );
    IO_print( out, "\t"                );
    IO_print( out, "OFFSET  "          );
    IO_print( out, dollar              );
    IO_print( out, "offset"            );
    IO_print( out, ";"                 );
    IO_print( out, "\n"                );
}

void
_outputQueryIfStatementWhereClauses( TableSpec* self, IO* out )
{
    bool primary_key_in_filter = false;

    const char*   _zero   = (PSQL == self->output) ? "'1970-01-01 00:00:00+00::timestamptz'" : "0";
    const char*   dollar  = (PSQL == self->output) ? "p_"                                     : "$";
    const String* filter  = (const String*) Dictionary_find_key  ( self->options, "Filter" );
    const String* prefix  = (const String*) Dictionary_find_key  ( self->options, "Prefix" );

    Array* parts = String_toArray_separator( filter, ';' );
    int    max   = prefix ? String_getLength( prefix ) + CharString_length( "deleted" ) : CharString_length( "deleted" );
    int    n     = Array_getCount( parts );

    max = CharString_length( _zero ) > max ? CharString_length( _zero ) : max; 

    for ( int i=0; i < n; i++ )
    {
        const String* part = (const String*) Array_get_index( parts, i );
              int     len  = String_getLength( part );

        if ( String_contentEquals( part, String_getContent( self->primaryKey ) ) )
        {
            primary_key_in_filter = true;
        }

        if ( max < len )
        {
            max = len;
        }
    }

    StringBuffer* sb = StringBuffer_new();
    if ( prefix )
    {
        StringBuffer_append_string( sb, prefix );
    }
    StringBuffer_append_chars ( sb, "deleted" ); 

    String* deleted = String_new_length( StringBuffer_getChars( sb ),           max );
    String* zero    = String_new_length( _zero,                                 max );
    String* primary = String_new_length( String_getContent( self->primaryKey ), max );

    IO_print( out, "\t"                                  );
    IO_print( out, "\t"                                  );
    IO_print( out, "WHERE   ("                           );
    IO_print( out, String_getContent( deleted )          );
    IO_print( out, " = "                                 );

    if ( !primary_key_in_filter )
    {
        IO_print( out, "0)"                              );
    }
    else
    {
        IO_print( out, String_getContent( zero )         );
        IO_print( out, "  OR "                           );
        IO_print( out, dollar                            );
        IO_print( out, String_getContent( primary )      );
        IO_print( out, "!= 0)"                           );
    }
    IO_print( out, "\n" );

    Delete( &sb      );
    Delete( &deleted );
    Delete( &zero    );
    Delete( &primary );

    for ( int i=0; i < n; i++ )
    {
        const String* part          = (const String*) Array_get_index( parts, i );
        const char*   default_value = "''";

        String* _part = String_new_length( String_getContent( part ), max );

        const Type* type = (const Type*) Dictionary_find_key( self->members, String_getContent( part ) );
        if ( type )
        {
            default_value = Type_defaultComparison( type, self->output );
        }

        IO_print( out, "\t"                       );
        IO_print( out, "\t"                       );
        IO_print( out, "AND     ("                );
        IO_print( out, String_getContent( _part ) );
        IO_print( out, " = "                      );
        IO_print( out, dollar                     );
        IO_print( out, String_getContent( _part ) );
        IO_print( out, " OR "                     );
        IO_print( out, dollar                     );
        IO_print( out, String_getContent( _part ) );
        IO_print( out, " = "                      );
        IO_print( out, default_value              );
        IO_print( out, ")"                        );
        IO_print( out, "\n"                       );

        Delete( &_part );
    }
}
