.. Type

The 'Type' class is used to store information about each field's type.
The 'name' member is the SQLGen type name -- 'Geo'.
The 'translated' member corresponds to the type name of the target database -- 'DECIMAL(10,7)'.
The 'defaultValue' member is the appropriate value for the member to be initialised with -- 0.0.
The 'foreign' member is true if the type represents a foreign key -- Int*.
The 'primary' member is true if the type represents a primary key -- Int \^{} .
The 'array' member is true if the type has a size -- 'Text[6]'.
The 'size' member is the specified size -- 6.

~source/include/sqlgen.h~
typedef struct _Type
{
    String*     name;
    String*     translated;
    String*     defaultValue;
    bool        foreign;
    bool        primary;
    bool        array;
    int         size;
} Type;
~

... Translate Name

~source/c/Type.c~
const char* Type_translateName( const Type* type, Output output, bool table )
{
    const char* ret = "?";

    const char* type_name = String_getContent( type->name );

    if ( CharString_contentEquals( type_name, "Blob" ) )
    {
        ret = "BLOB";
    }
    else
    if ( CharString_contentEquals( type_name, "Bool" ) )
    {
        ret = "BOOL";
    }
    else
    if ( CharString_contentEquals( type_name, "Boolean" ) )
    {
        ret = "BOOL";
    }
    else
    if ( CharString_contentEquals( type_name, "Char" ) )
    {
        ret = "CHAR";
    }
    else
    if ( CharString_contentEquals( type_name, "Coordinate" ) )
    {
        ret = "DECIMAL(10,7)";
    }
    else
    if ( CharString_contentEquals( type_name, "Date" ) )
    {
        ret = "DATE";
    }
    else
    if ( CharString_contentEquals( type_name, "DateTime" ) )
    {
        switch ( output )
        {
        case PSQL:
            ret = "TIMESTAMP";
            break;

        default:
            ret = "DATETIME";
        }
    }
    else
    if ( CharString_contentEquals( type_name, "Datetime" ) )
    {
        switch ( output )
        {
        case PSQL:
            ret = "TIMESTAMP";
            break;

        default:
            ret = "DATETIME";
        }
    }
    else
    if ( CharString_contentEquals( type_name, "Double" ) )
    {
        switch ( output )
        {
        case PSQL:
            ret = "DOUBLE PRECISION";
            break;

        default:
            ret = "DOUBLE";
        }
    }
    else
    if ( CharString_contentEquals( type_name, "Float" ) )
    {
        switch ( output )
        {
        case PSQL:
            ret = "REAL";
            break;

        default:
            ret = "FLOAT";
        }
    }
    else
    if ( CharString_contentEquals( type_name, "Geo" ) )
    {
        ret = "DECIMAL(10,7)";
    }
    else
    if ( CharString_contentEquals( type_name, "GUID" ) )
    {
        ret = "CHAR(36)";
    }
    else
    if ( CharString_contentEquals( type_name, "Guid" ) )
    {
        ret = "CHAR(36)";
    }
    else
    if ( CharString_contentEquals( type_name, "Int" ) )
    {
        ret = "INT";

        if ( type->primary )
        {
            switch( output )
            {
            case PSQL:
                if ( table )
                {
                    ret = "SERIAL";
                }
                else
                {
                    ret = "INT";
                }
                break;

            default:
                ret = "INT";
            }
        }
    }
    else
    if ( CharString_contentEquals( type_name, "Integer" ) )
    {
        ret = "INT";
    }
    else
    if ( CharString_contentEquals( type_name, "JSON" ) )
    {
        ret = "JSON";
    }
    else
    if ( CharString_contentEquals( type_name, "LongBlob" ) )
    {
        ret = "LONGBLOB";
    }
    else
    if ( CharString_contentEquals( type_name, "LongText" ) )
    {
        switch( output )
        {
        case PSQL:
            ret = "TEXT";
            break;

        default:
            ret = "LONGTEXT";
        }
    }
    else
    if ( CharString_contentEquals( type_name, "Longtext" ) )
    {
        switch( output )
        {
        case PSQL:
            ret = "TEXT";
            break;

        default:
            ret = "LONGTEXT";
        }
    }
    else
    if ( CharString_contentEquals( type_name, "MediumText" ) )
    {
        switch( output )
        {
        case PSQL:
            ret = "TEXT";
            break;

        default:
            ret = "MEDIUMTEXT";
        }
    }
    else
    if ( CharString_contentEquals( type_name, "Mediumtext" ) )
    {
        switch( output )
        {
        case PSQL:
            ret = "TEXT";
            break;

        default:
            ret = "MEDIUMTEXT";
        }
    }
    else
    if ( CharString_contentEquals( type_name, "Money" ) )
    {
        ret = "DECIMAL(11,2)";
    }
    else
    if ( CharString_contentEquals( type_name, "Text" ) )
    {
        ret = type->size ? "TEXT" : "TEXT";
    }
    else
    if ( CharString_contentEquals( type_name, "Time" ) )
    {
        ret = "TIME";
    }
    else
    if ( CharString_contentEquals( type_name, "TimeStamp" ) )
    {
        switch( output )
        {
        case PSQL:
            ret = "TIMESTAMPTZ";
            break;

        default:
            ret = "TIMESTAMP";
        }
    }
    else
    if ( CharString_contentEquals( type_name, "Timestamp" ) )
    {
        switch( output )
        {
        case PSQL:
            ret = "TIMESTAMPTZ";
            break;

        default:
            ret = "TIMESTAMP";
        }
    }
    else
    if ( CharString_contentEquals( type_name, "Year" ) )
    {
        ret = "YEAR";
    }
    return ret;
}
