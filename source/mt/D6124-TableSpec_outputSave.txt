DELIMITER //
CREATE PROCEDURE spgen_<table name>_save
(
    $<primary>  INT,
    $field      TEXT,
    $value      TEXT
)
BEGIN

    CASE $field
    WHEN $field = <Save> THEN UPDATE <table name> SET <Save> = $value WHERE <primary> = $<primary;
    ELSE

        THROW

    END CASE;

END
//
DELIMTIER ;

void
_outputSave( TableSpec* self, IO* out )
{
    const String* save = (const String*) Dictionary_find_key( self->options, "Save" );

    if ( save )
    {
        String* primary = null;

        const Array* entries = Dictionary_getEntries( self->members );
              int    n       = Array_getCount( entries );



        for ( int i=0; i < n; i++ )
        {
            const Entry*  entry = (const Entry*) Array_get_index( entries, i );
            const String* name  =                Entry_getKey   ( entry      );
            const Type*   type  = (const Type*)  Entry_getValue ( entry      );

            if ( type->primary )
            {
                primary = String_new( String_getContent( name ) );
            }
        }

        if ( primary )
        {
            const char* _primary = String_getContent( primary );

            IO_print( out, "DELIMITER //"                  );
            IO_print( out, "\n" );

            IO_print( out, "CREATE PROCEDURE "             );
            IO_print( out, "spgen_"                        );
            IO_print( out, String_getChars( self->name )   );
            IO_print( out, "_save"                         );
            IO_print( out, "\n"                            );
            IO_print( out, "("                             );
            IO_print( out, "\n"                            );

            _outputSaveParameters( self, out );

            IO_print( out, ")"                             );
            IO_print( out, "\n"                            );

            IO_print( out, "SQL SECURITY INVOKER"          );
            IO_print( out, "\n"                            );
            IO_print( out, "COMMENT 'Generated by sqlgen'" );
            IO_print( out, "\n"                            );
            IO_print( out, "BEGIN"                         );
            IO_print( out, "\n"                            );
            IO_print( out, "\n"                            );

            IO_print( out, "\t"     );
            IO_print( out, "IF "    );
            IO_print( out, "$"      );
            IO_print( out, _primary );
            IO_print( out, " THEN"  );
            IO_print( out, "\n"     );
            IO_print( out, "\n"     );

            IO_print( out, "\t" );
            IO_print( out, "\t" );
            IO_print( out, "CASE $name" );
            IO_print( out, "\n" );

            _outputSaveWhenStatements( self, out );

            IO_print( out, "\t"   );
            IO_print( out, "\t"   );
            IO_print( out, "ELSE" );
            IO_print( out, "\n"   );
            IO_print( out, "\n"   );

            IO_print( out, "\t" );
            IO_print( out, "\t" );
            IO_print( out, "\t" );

            IO_print( out, "SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'INVALID CASE OPTION IN " );
            IO_print( out, "spgen_" );
            IO_print( out, String_getContent( self->name ) );
            IO_print( out, "_save" );
            IO_print( out, "';" );
            IO_print( out, "\n"                            );
            IO_print( out, "\n"                            );

            IO_print( out, "\t" );
            IO_print( out, "\t" );
            IO_print( out, "END CASE;" );
            IO_print( out, "\n"                            );
            IO_print( out, "\n"                            );

            IO_print( out, "\t"      );
            IO_print( out, "END IF;" );
            IO_print( out, "\n"                            );
            IO_print( out, "\n"                            );

            IO_print( out, "END"                           );
            IO_print( out, "\n"                            );
            IO_print( out, "//"                            );
            IO_print( out, "\n"                            );
            IO_print( out, "DELIMITER ;"                   );
            IO_print( out, "\n"                            );
        }
        Delete( &primary );
    }
}

void
_outputSaveParameters( TableSpec* self, IO* out )
{
    const Array* entries = Dictionary_getEntries( self->members );
          int    max     = 5;
          int    n       = Array_getCount( entries );

    for ( int i=0; i < n; i++ )
    {
        const Entry*  entry = (const Entry*) Array_get_index( entries, i );
        const String* name  =                Entry_getKey   ( entry      );
        const Type*   type  = (const Type*)  Entry_getValue ( entry      );
              int     len   =                String_getLength( name );

        if ( type->foreign || type->primary )
        {
            max = (len > max) ? len : max;
        }
    }

    String* field = String_new_length( "name",  max );
    String* value = String_new_length( "value", max );
    {
        for ( int i=0; i < n; i++ )
        {
            const Entry*  entry = (const Entry*) Array_get_index( entries, i );
            const String* name  =                Entry_getKey   ( entry      );
            const Type*   type  = (const Type*)  Entry_getValue ( entry      );

            if ( type->foreign || type->primary )
            {
                String* _name = String_new_length( String_getContent( name ), max );
                {
                    IO_print( out, "\t"                       );
                    IO_print( out, "$"                        );
                    IO_print( out, String_getContent( _name ) );
                    IO_print( out, " "                        );
                }
                Delete( &_name );

                if ( type && type->translated )
                {
                    IO_print( out, String_getContent( type->translated ) );
                }
                IO_print( out, ","                        );
                IO_print( out, "\n"                       );
            }
        }

        IO_print( out, "\t"                           );
        IO_print( out, "$"                            );
        IO_print( out, String_getContent( field )     );
        IO_print( out, " "                            );
        IO_print( out, "TEXT"                         );
        IO_print( out, ","                            );
        IO_print( out, "\n"                           );

        IO_print( out, "\t"                           );
        IO_print( out, "$"                            );
        IO_print( out, String_getContent( value )     );
        IO_print( out, " "                            );
        IO_print( out, "TEXT"                         );
        IO_print( out, "\n"                           );
    }
    Delete( &field     );
    Delete( &value     );
}

~source/c/TableSpec.c~
void
_outputSaveWhenStatements( TableSpec* self, IO* out )
{
    const String* prefix = (const String*) Dictionary_find_key( self->options, "Prefix" );
    const String* save   = (const String*) Dictionary_find_key( self->options, "Save"   );

    if ( prefix && save && self->primaryKey )
    {
        Array*  parts = String_toArray_separator( save, ';' );
        int     n     = Array_getCount( parts );
        int     max   = 0;

        for ( int i=0; i < n; i++ )
        {
            const String* save_name = (const String*)  Array_get_index( parts, i );

            int len = String_getLength( save_name );

            max = (max < len) ? len : max;
        }

        const char* _primary = String_getContent( self->primaryKey );

        for ( int i=0; i < n; i++ )
        {
            const String* save_name = (const String*)  Array_get_index( parts, i );

            if ( save_name )
            {
                      String*  quoted        = String_new_format( "'%s'", String_getContent( save_name ) );
                      String*  quoted_padded = String_new_length( String_getContent( quoted ), max + 2   );
                const char*   _quoted_padded = String_getContent( quoted_padded                          );
                      String*  padded        = String_new_length( String_getContent( save_name ), max    );
                const char*   _padded        = String_getContent( padded                                 );
                const char*   _prefix        = String_getContent( prefix                                 );

                IO_print( out, "\t"               );
                IO_print( out, "\t"               );
                IO_print( out, "WHEN "            );
                IO_print( out, _quoted_padded     );
                IO_print( out, "THEN UPDATE "     );

                IO_print( out, String_getContent( self->name ) );

                IO_print( out, " SET "            );

                IO_print( out, _padded            );
                IO_print( out, " = $value "       );
                IO_print( out, "WHERE "           );
                IO_print( out, _prefix            );
                IO_print( out, "deleted = 0"      );
                IO_print( out, " AND "            );
                IO_print( out, _primary           );
                IO_print( out, " = "              );
                IO_print( out, "$"                );
                IO_print( out, _primary           );
                IO_print( out, ";"                );
                IO_print( out, "\n"               );
            }
        }

        Delete( &parts );
    }
}
