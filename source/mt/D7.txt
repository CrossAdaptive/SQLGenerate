static void populate_typemap_mysql()
{
    typemap[MYSQL][Blob]       = _BLOB;
    typemap[MYSQL][Bool]       = _BOOL;
    typemap[MYSQL][Boolean]    = _BOOLEAN;
    typemap[MYSQL][Char]       = _CHAR;
    typemap[MYSQL][Coordinate] = _DECIMAL307;
    typemap[MYSQL][Date]       = _DATE;
    typemap[MYSQL][Datetime]   = _DATETIME;
    typemap[MYSQL][Double]     = _DOUBLE;
    typemap[MYSQL][Float]      = _FLOAT;
    typemap[MYSQL][Geo]        = _GEO;
    typemap[MYSQL][Guid]       = _CHAR36;
    typemap[MYSQL][Int]        = _INT;
    typemap[MYSQL][Integer]    = _INT;

    typemap[MYSQL][Longblob]   = _LONGBLOB;
