~source/c/TableSpec.c~
void
TableSpec_output( TableSpec* self, IO* out )
{
    if ( self->valid )
    {
        IO_setTabspace( out, true );
        _outputSQL( self, out );
        IO_flushAll();
    }
}
~

~source/c/TableSpec.c~
void
_outputSQL( TableSpec* self, IO* out )
{
    if ( self->database )
    {
        _outputView( self, out );
    }
    else
    {
        _outputTable( self, out );
    }

    _outputQuery  ( self, out );
    //_outputReplace( self, out );
    //_outputSave   ( self, out );
    //_outputDelete ( self, out );

}
