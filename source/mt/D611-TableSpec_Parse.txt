bool _printError( const Token* token, IO* out, Tokenizer* tokenizer, const char* expected, const char* message )
{
    if ( true )
    {
        Console_Colour( COLOR_WARNING );
        _printToken( token, out );
        Console_Colour( COLOR_LIGHT );
        {
            Token* token;
            bool   loop = true;

            while ( loop && (token = Tokenizer_nextToken( tokenizer )) )
            {
                _printToken( token, out );

                if ( TOKEN_CLOSE == Token_getType( token ) )
                {
                    if ( String_contentEquals( Token_getContent( token ), "}" ) )
                    {
                        loop = false;
                    }
                }
                Delete( &token );
            }

            token = Tokenizer_nextRealToken( tokenizer, null );
            if ( Token_contentEquals_chars( token, "\n" ) )
            {
                _printToken( token, out );
            }
            else
            {
                Tokenizer_pushBack_token( tokenizer, &token ); 
            }
            Delete( &token );
        }
    }

    const char* token_value = String_getChars( Token_getContent( token ) );
    const char* token_type  = Token_getTypeName( token );
          int   line        = Token_getLine    ( token );
          int   column      = Token_getColumn  ( token );

    IO_print( out, "\n" );
    IO_print( out, "\n" );

    Console_Colour( COLOR_NORMAL  );
    Console_Colour( COLOR_WARNING );
    Console_Print_pattern_valist( "Error: Expected %s %s %s (%s) at line: %i, column: %i\n", expected, message, token_value, token_type, line, column );
    Console_Colour( COLOR_LIGHT );

    return false;
}

static String* _transformToLowerCaseString( const char* content )
{
    String*       ret = null;
    StringBuffer* sb  = StringBuffer_new();
    char          ch;

    while ( *content )
    {
        ch = tolower( (unsigned char) *content );

        StringBuffer_append_char( sb, ch );
        content++;
    }

    ret = String_new( StringBuffer_getChars( sb ) );

    Delete( &sb );

    return ret;
}

TableSpec*
TableSpec_Parse( Tokenizer* tokenizer, IO* out, const String* output, const String* database )
{
    TableSpec* self      = TableSpec_new( output, database );
    Token*     token     = null;
    bool       loop      = true;

    while ( loop && (token = Tokenizer_nextRealToken( tokenizer, null )) )
    {
        if ( TOKEN_COMMENT == Token_getType( token ) )
        {
            //  ignore comment
        }
        else
        if ( TOKEN_NEWLINE == Token_getType( token ) )
        {
            //  ignore newline
        }
        else
        if ( !Token_contentEquals_chars( token, "table" ) )
        {
            loop = _printError( token, out, tokenizer, "table", "Invalid token:" );
        }
        else
        {
            Console_Colour( COLOR_COMMENT );
            IO_print( out, "#\n"     );
            IO_print( out, "# table" );

            self->valid = _parseTableName( self, tokenizer, out );

            if ( self->valid )
            {
                IO_print( out, "\n#\n" );
            }
            Console_Colour( COLOR_NORMAL );
            loop = false;
        }

        Delete( &token );
    }

    if ( !self->valid )
    {
        Delete( &self->members );
        Delete( &self->options );
        Delete( &self          );
    }
   
    return self;
}

bool _parseTableInner( TableSpec* self, Tokenizer* tokenizer, IO* out )
{
    bool   ret   = false;
    bool   loop  = true;
    Token* token = null;

    while ( loop && (token = Tokenizer_nextRealToken( tokenizer, out )) )
    {
        switch ( Token_getType( token ) )
        {
        case TOKEN_COMMENT:
        case TOKEN_NEWLINE:
            _printToken( token, out );
            //  ignore comment
            break;

        case TOKEN_CLOSE:
            if ( !Token_contentEquals_chars( token, "}" ) )
            {
                loop = _printError( token, out, tokenizer, "@,#,}", "Invalid token:" );
            }
            else
            {
                _printToken( token, out );
                ret = true;
            }
            loop = false;
            break;

        case TOKEN_SYMBOL_PREFIX:
        case TOKEN_SYMBOL_CONTEXTUAL:
        case TOKEN_SYMBOL:
            if ( Token_contentEquals_chars( token, "@" ) )
            {
                _printToken( token, out );
                loop = (ret = _parseTableInnerMember( self, tokenizer, out ));
            }
            else
            if ( Token_contentEquals_chars( token, "%" ) )
            {
                _printToken( token, out );
                loop = (ret = _parseTableInnerOption( self, tokenizer, out ));
            }
            else
            {
                loop = _printError( token, out, tokenizer, "@,#,}", "Invalid token:" );
            }
            break;

        default:
            loop = _printError( token, out, tokenizer, "@,#,}", "Invalid token:" );
        }
    }
    return ret;
}

.... parse Table Inner Member Type Open

~
//  
//  table base_users { @trident: Text }
//  table base_users { @trident: Text @next...
//  table base_users { @trident: Text %next...
//  table base_users { @trident: Text [10] }
//                                    ^
~

~source/c/TableSpec.c~
Type* _parseTableInnerMemberTypeOpen( TableSpec* self, Tokenizer* tokenizer, IO* out )
{
    Type*   ret   = null;
    Token*  token = null;
    bool    loop  = true;

    while ( loop && (token = Tokenizer_nextRealToken( tokenizer, out )) )
    {
        switch ( Token_getType( token ) )
        {
        case TOKEN_COMMENT:
        case TOKEN_NEWLINE:
            _printToken( token, out );
            //  ignore comment
            break;

        case TOKEN_SYMBOL_PREFIX:
        case TOKEN_SYMBOL_CONTEXTUAL:
        case TOKEN_SYMBOL:
            {
            const char* content = String_getContent( Token_getContent( token ) );
            if
            (
                ('@' == content[0]) ||
                ('%' == content[0]) ||
                ('}' == content[0])
            )
            {
                Tokenizer_pushBack_token( tokenizer, &token );

                ret = New( Type );
            }
            else
            if ( Token_contentEquals_chars( token, "^" ) )
            {
                _printToken( token, out );
                ret          = New( Type );
                ret->primary = true;
            }
            else
            if ( Token_contentEquals_chars( token, "*" ) )
            {
                _printToken( token, out );
                ret          = New( Type );
                ret->foreign = true;
            }
            else
            {
                loop = _printError( token, out, tokenizer, ":", "Invalid token:" );
            }
            loop = false;
            }
            break;

        case TOKEN_OPEN:
            _printToken( token, out );
            if ( !Token_contentEquals_chars( token, "[" ) )
            {
                loop = _printError( token, out, tokenizer, ":", "Invalid token:" );
            }
            else
            {
                int size = _parseTableInnerMemberTypeSize( self, tokenizer, out );

                if ( 0 <= size )
                {
                    ret        = New( Type );
                    ret->array = true;
                    ret->size  = size;
                }
                loop = false;
            }
            break;

        default:
            _printToken( token, out );
            loop = _printError( token, out, tokenizer, "<value>", "Invalid token:" );
        }
    }
    return ret;
}

bool _parseTableInnerOption( TableSpec* self, Tokenizer* tokenizer, IO* out )
{
    bool   ret   = false;
    bool   loop  = true;
    Token* token = null;

    String* name  = null;
    String* value = null;

    while ( loop && (token = Tokenizer_nextRealToken( tokenizer, out )) )
    {
        switch ( Token_getType( token ) )
        {
        case TOKEN_COMMENT:
        case TOKEN_NEWLINE:
            _printToken( token, out );
            //  ignore comment
            break;

        case TOKEN_WORD:
            name  =  String_new( String_getContent( Token_getContent( token ) ) );
            if
            (
                String_contentEquals( name, "Filter"  ) ||
                String_contentEquals( name, "OrderBy" ) ||
                String_contentEquals( name, "Save"    )
            )
            {
                _printToken( token, out );
                value = _parseTableInnerOptionSep( self, tokenizer, out, true );
            }
            else
            if
            (
                String_contentEquals( name, "Module"   ) ||
                String_contentEquals( name, "Output"   ) ||
                String_contentEquals( name, "Prefix"   ) ||
                String_contentEquals( name, "Security" ) ||
                String_contentEquals( name, "View"     )
            )
            {
                _printToken( token, out );
                value = _parseTableInnerOptionSep( self, tokenizer, out, false );
            }
            else
            {
                loop = _printError( token, out, tokenizer, "<name>", "Invalid token:" );
            }
            ret   = (null != value);
            loop  = false;
            break;

        default:
            loop = _printError( token, out, tokenizer, "<name>", "Invalid token:" );
        }
    }

    if ( name && value )
    {
        const char* key = String_getContent( name );

        Dictionary_add_key_object( self->options, key, (Object**) &value );

        Delete( &name );
    }

    return ret;
}

//  table base_users { %OrderBy = ONE @next...

const char* _parseTableInnerOptionValueSep( TableSpec* self, Tokenizer* tokenizer, IO* out, bool is_list )
{
    const char* ret   = null;
    Token*      token = null;
    bool        loop  = true;

    while ( loop && (token = Tokenizer_nextRealToken( tokenizer, out )) )
    {
        switch ( Token_getType( token ) )
        {
        case TOKEN_COMMENT:
        case TOKEN_NEWLINE:
            _printToken( token, out );
            //  ignore comment
            break;

        case TOKEN_SYMBOL_PREORPOSTFIX:
            if ( Token_contentEquals_chars( token, "+" ) )
            {
                _printToken( token, out );
                ret  = " ";
                loop = false;
            }
            else
            {
                loop = _printError( token, out, tokenizer, "<value>", "Invalid token:" );
            }
            break;

        case TOKEN_CLOSE:
        case TOKEN_SYMBOL_CONTEXTUAL:
        case TOKEN_SYMBOL:
            if
            (
                Token_contentEquals_chars( token, "}" ) ||
                Token_contentEquals_chars( token, "%" )
            )
            {
                Tokenizer_pushBack_token( tokenizer, &token );
                ret  = "";
                loop = false;
            }
            else
            if ( Token_contentEquals_chars( token, "," ) )
            {
                _printToken( token, out );
                ret  = ",";
                loop = false;
            }
            else
            if ( is_list && Token_contentEquals_chars( token, ";" ) )
            {
                _printToken( token, out );
                ret  = ";";
                loop = false;
            }
            else
            {
                loop = _printError( token, out, tokenizer, "<value>", "Invalid token:" );
            }
            break;

        default:
            loop = _printError( token, out, tokenizer, "<value>", "Invalid token:" );
        }
        Delete( &token );
    }
    return ret;
}
