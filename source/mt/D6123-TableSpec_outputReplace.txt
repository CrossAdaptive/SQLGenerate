void
_outputReplace( TableSpec* self, IO* out )
{
    IO_print( out, "DELIMITER //"                  );
    IO_print( out, "\n" );

    IO_print( out, "CREATE PROCEDURE "             );
    IO_print( out, "spgen_"                        );
    IO_print( out, String_getChars( self->name )   );
    IO_print( out, "_replace"                      );
    IO_print( out, "\n"                            );
    IO_print( out, "("                             );
    IO_print( out, "\n"                            );

    _outputReplaceParameters( self, out );

    IO_print( out, ")"                             );
    IO_print( out, "\n"                            );

    IO_print( out, "SQL SECURITY INVOKER"          );
    IO_print( out, "\n"                            );
    IO_print( out, "COMMENT 'Generated by sqlgen'" );
    IO_print( out, "\n"                            );
    IO_print( out, "BEGIN"                         );
    IO_print( out, "\n"                            );
    IO_print( out, "\n"                            );

/*

    IF ISNULL($PRIMARY) OR 0 = $PRIMARY THEN
        DO LAST_INSERT_ID( 0 );
    ELSE
        DO LAST_INSERT_ID( $PRIMARY );
    END IF;

    INSERT INTO <table name>
    (
        fields.
        ...
    )
    VALUES
    (
        $fields
    )
    ON DUPLICATE KEY UPDATE
        field = IFNULL( $field, field ),
        ...
        PRIMARY = LAST_INSERT_ID( PRIMARY )
    ;
*/

    IO_print( out, "\t" );
    IO_print( out, "IF ISNULL($" );
    IO_print( out, String_getContent( self->primaryKey ) );
    IO_print( out, ") OR 0 = $" );
    IO_print( out, String_getContent( self->primaryKey ) );
    IO_print( out, " THEN" );
    IO_print( out, "\n" );
    IO_print( out, "\t" );
    IO_print( out, "\t" );
    IO_print( out, "DO LAST_INSERT_ID(0);" );
    IO_print( out, "\n" );
    IO_print( out, "\t" );
    IO_print( out, "ELSE" );
    IO_print( out, "\n" );
    IO_print( out, "\t" );
    IO_print( out, "\t" );
    IO_print( out, "DO LAST_INSERT_ID($" );
    IO_print( out, String_getContent( self->primaryKey ) );
    IO_print( out, ");" );
    IO_print( out, "\n" );
    IO_print( out, "\t" );
    IO_print( out, "END IF;" );
    IO_print( out, "\n" );
    IO_print( out, "\n" );

    IO_print( out, "\t" );
    IO_print( out, "INSERT INTO " );
    IO_print( out, String_getContent( self->name ) );
    IO_print( out, "\n" );
    IO_print( out, "\t" );
    IO_print( out, "(" );
    IO_print( out, "\n" );

    _outputReplaceFields( self, out );

    IO_print( out, "\t" );
    IO_print( out, ")" );
    IO_print( out, "\t" );
    IO_print( out, "\n" );

    IO_print( out, "\t" );
    IO_print( out, "VALUES" );
    IO_print( out, "\n" );
    IO_print( out, "\t" );
    IO_print( out, "(" );
    IO_print( out, "\n" );

    _outputReplaceValues( self, out );

    IO_print( out, "\t" );
    IO_print( out, ")" );
    IO_print( out, "\n" );
    IO_print( out, "\t" );
    IO_print( out, "ON DUPLICATE KEY UPDATE" );
    IO_print( out, "\n" );

    _outputReplaceUpdate( self, out );

    IO_print( out, "\t" );
    IO_print( out, ";" );
    IO_print( out, "\n" );

    IO_print( out, "\t" );
    IO_print( out, "SET " );
    IO_print( out, "$" );
    IO_print( out, String_getContent( self->primaryKey ) );
    IO_print( out, " = LAST_INSERT_ID();" );
    IO_print( out, "\n" );

    IO_print( out, "END"                           );
    IO_print( out, "\n"                            );
    IO_print( out, "//"                            );
    IO_print( out, "\n"                            );
    IO_print( out, "DELIMITER ;"                   );
    IO_print( out, "\n"                            );

    IO_flushAll();
}

void
_outputReplaceParameters( TableSpec* self, IO* out )
{
    const Array* entries = Dictionary_getEntries( self->members );
          int    max     = 0;
          int    n       = Array_getCount( entries );

    for ( int i=0; i < n; i++ )
    {
        const Entry*  entry = (const Entry*) Array_get_index( entries, i );
        const String* name  = Entry_getKey( entry );
              int     len   = String_getLength( name );

        if ( max < len )
        {
            max = len;
        }
    }

    for ( int i=0; i < n; i++ )
    {
        const Entry*  entry = (const Entry*)  Array_get_index( entries, i );
        const String* name  = (const String*) Entry_getKey   ( entry      );
        const Type*   type  = (const Type*)   Entry_getValue ( entry      );

        if ( type->primary )
        {
            IO_print( out, "INOUT"                    );
            IO_print( out, "\n"                       );
        }

        String* _name = String_new_length( String_getContent( name ), max );
        {
            IO_print( out, "\t"                       );
            IO_print( out, "$"                        );
            IO_print( out, String_getContent( _name ) );
            IO_print( out, " "                        );
        }

        if ( type && type->translated )
        {
            IO_print( out, String_getContent( type->translated ) );
        }

        if ( i < (n - 1) )
        {
            IO_print( out, "," );
        }
        IO_print( out, "\n" );
    }
}

void
_outputReplaceUpdate( TableSpec* self, IO* out )
{
    const Array* entries = Dictionary_getEntries( self->members );
          int    max     = 0;
          int    n       = Array_getCount( entries );

    for ( int i=0; i < n; i++ )
    {
        const Entry*  entry = (const Entry*) Array_get_index( entries, i );
        const String* name  = Entry_getKey( entry );
              int     len   = String_getLength( name );

        if ( max < len )
        {
            max = len;
        }
    }

    String*  primary = null;
    String* _primary = null;

    for ( int i=0; i < n; i++ )
    {
        const Entry*  entry = (const Entry*)  Array_get_index( entries, i );
        const String* name  = (const String*) Entry_getKey   ( entry      );
        const Type*   type  = (const Type*)   Entry_getValue ( entry      );

        if ( type->primary )
        {
             primary = String_new       ( String_getContent( name )      );
            _primary = String_new_length( String_getContent( name ), max );
        }
        else
        if ( type->foreign )
        {
            // Skip foreign keys for update.
        }
        else
        {
            //String* _name = String_new( String_getContent( name ) );
            String* _name = String_new_length( String_getContent( name ), max );
            {
                IO_print( out, "\t"                       );
                IO_print( out, "\t"                       );
                IO_print( out, String_getContent( _name ) );
                IO_print( out, " = IFNULL("               );
                IO_print( out, "$"                        );
                IO_print( out, String_getContent( _name ) );
                IO_print( out, ", "                       );
                IO_print( out, String_getContent( _name ) );
                IO_print( out, "),"                       );
                IO_print( out, "\n"                       );
            }
            Delete( &_name );
        }
    }

    if ( primary )
    {
        IO_print( out, "\t"                          );
        IO_print( out, "\t"                          );
        IO_print( out, String_getContent( _primary ) );
        IO_print( out, " = LAST_INSERT_ID("          );
        IO_print( out, String_getContent(  primary ) );
        IO_print( out, ")"                           );
        IO_print( out, "\n"                          );
    }
    Delete( & primary );
    Delete( &_primary );
}
