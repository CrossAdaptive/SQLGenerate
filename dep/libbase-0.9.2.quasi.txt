. Introduction

. Example Usage

The following shows an example usage of the 'libbase' library.
For a guide to compling a program that uses 'libbase' please see Appendix A.
To use the functions associated with 'libase' simply include 'libbase.h'.

~!source/c/main.c~
#include "libbase.h"
~

~source/c/main.c~
const char* flags[]
=
{
    "-v",            "--version",
    "-h",            "--help",
    "-o=<Out file>", "--out=<Out file>",
    "...",
    null
};
~

~source/c/main.c~
int main( int argc, const char** argv )
{
    Arguments* arguments = Arguments_new_count_arguments_flags( argc, argv, flags );

    Console_Print_pattern_valist( "Objects(%i) vs Arrays(%i)\n", Platform_Objects(), Platform_Arrays() );

    Array* strings = Array_new();

    String* one = String_new( "one" );

    if ( true )
    {
        Console_Print_pattern_valist( "Objects(%i) vs Arrays(%i)\n", Platform_Objects(), Platform_Arrays() );

        Array_append_element( strings, Platform_Give( String_new( "Hello" ) ) );
        Array_append_element( strings, Platform_Give( String_new( "World" ) ) );

        Console_Print_pattern_valist
        (
            "%s %s\n",
            String_getChars( Array_get_index( strings, 0 ) ),
            String_getChars( Array_get_index( strings, 1 ) )
        );
    }
    Delete( &one );

    Delete( &strings );

    Console_Print_pattern_valist( "Objects(%i) vs Arrays(%i)\n", Platform_Objects(), Platform_Arrays() );
}
~

. Types

!
~!include/libbase.h~
#ifndef LIBBASE_H
#define LIBBASE_H

#include "libbase/_types.h"
~
!

The file '_types.h' is implicitly included when a source file includes 'libbase.h'.
The main purpose of the file is to define various macros and types.
The macros are used to facilitate memory usage.
They types that defined are the 'Destructor' function type that is used to deallocate an object of a specific class;
and the object types themselves.

~!include/libbase/_types.h~
#ifndef LIBBASE_TYPES_H
#define LIBBASE_TYPES_H
~

.. The OBJECT magic number

~include/libbase/_types.h~
#define OBJECT                  0xFAFAFAFA
~

Each object has an integer value as it's first member that is set to the value of the 'OBJECT' magic number.
This allows the memory management functions to determine if a memory pointer points to an object that should be deallocated using a destructor, or whether it can be freed directly using 'free'.

.. The 'New' and 'NewArray' macros

~include/libbase/_types.h~
#define New( Type )             Platform_Alloc     ( sizeof(Type), 0       )
#define NewArray( Type, count ) Platform_Alloc     ( sizeof(Type), count   )
~

The 'New' and 'NewArray' macros are convenience macros that allow memory to be allocated by specifying a type name.
Instead of calling 'sizeof', code can simply call:

~
MyObject* self = New( MyObject );
~

or in the case of an array of 'MyObject' objects:

~
MyObject** self = NewArray( MyObject, 100 );
~

It should be noted that the 'New' macro would normally be used in a class constructor to allocate memory of an instantiated object.
Typically,
the allocation of a normal object would be achieved by calling its '<Classs name>_new' function.

Similarly, the 'NewArray' macro should only be rarely used where it makes sense.
Normally,
user code should use the 'Array' class,
which provides more memory safety than managing memory oneself,
or should create an 'ArrayOf<Class>' class that wraps array.

/Note: yes 'Platform_Alloc' takes '0' as the argument when allocating memory for one object./
/This allows it to differentiate between the allocation of single objects and arrays -- which are tracked in static counters./

.. The Delete and DeleteArray macros

~include/libbase/_types.h~
#define Delete( element )       Platform_Delete   ( (void**) element )
#define DeleteArray( element )  Platform_FreeArray( (void**) element )
#define Give( element )         Platform_Give     ( (void**) element )
#define Take( element )         Platform_Take     ( (void**) element )
~

As discussed in the previous chapter,
'libbase' uses the convention of passing object "ownership" using memory pointer-to-pointers.
This allows the receiving function to copy the pointer value and then nullify the existing value the caller has access to.

Unfortunately,
while C allows any pointer type to be passed to function that accepts 'void*',
the same is not true for pointer-to-pointer types.
The 'Delete' and 'DeleteArray' macros simply cast any value to the 'void**' type.
This means, however, that the compiler is not able to type check what is being passed.

.. Defined values

~include/libbase/_types.h~
#define null    0
#define false   0
#define true    1
~

.. The 'Destructor' function pointer

~include/libbase/_types.h~
typedef void(*Destructor)(void*);
~

For each object,
a destructor function pointer is stored in the base 'Object' that is called by 'Platform_Delete' function if the passed value is an object.
Such destructor functions are responsible for also calling any parent object destructors.

void* Platform_Alloc( int size, int count )
{
    if ( 0 == count )
    {
        _Platform.objects++;
        return calloc( 1, size );
    }
    else
    {
        _Platform.arrays++;
        return calloc( count, size );
    }

... Platform Delete

~source/c/Platform.c~
void*
Platform_Delete( void** element )
{
    if ( element && *element )
    {
        Object* obj = *element;

        switch ( obj->type )
        {
        case OBJECT:
            obj->destruct( *element );
            break;
        }
        free( *element ); *element = 0;
        _Platform.objects--;
    }
    return null;
}
~

... Function: Array_destruct

~source/c/Array.c~
void
Array_destruct( Array* self )
{
    if ( self )
    {
        for ( int i=0; i < self->capacity; i++ )
        {
            if ( self->elements[i] )
            {
                Delete( &(self->elements[i]) );
            }
        }
        DeleteArray( &self->elements );
        self->count    = 0;
        self->capacity = 0;
        Object_destruct( &self->super );
    }
}
~

... Function: Array_new

~source/c/Array.c~
Array*
Array_new()
{
    Array* self = null;
    if ( (self = self ? self : New( Array )) )
    {
        Object_super( &self->super, (Destructor) Array_destruct );

        self->count    = 0;
        self->capacity = DefaultCapacity;
        self->elements = NewArray( void*, self->capacity );
        self->first    = -1;
        self->last     = -1;
    }
    return self;
}

... Function: Array_append_element

~source/c/Array.c~
void
Array_append_element( Array* self, void** element )
{
    if ( element && *element )
    {
        int index = Array_getLastIndex( self ) + 1;

        Array_resize( self, index );

        self->elements[index] = *element; *element = 0;
        self->last            = index;
        self->count++;

        if ( -1 == self->first ) self->first = index;
    }
}
~

... Function: Array_replace_element_index

~source/c/Array.c~
void*
Array_replace_element_index( Array* self, int index, void** element )
{
    void* ret = self->elements[index];

    if ( null == ret )
    {
        self->count++;
    }

    if ( null == *element )
    {
        self->count--;
    }

    self->elements[index] = *element; *element = null;

    if ( -1 == self->first || index < self->first )
    {
        self->first = index;
    }

    if ( -1 == self->last || index > self->last )
    {
        self->last = index;
    }

    return ret;
}
~

\newpage

... Function: Array_remove_index

~source/c/Array.c~
void*
Array_remove_index( Array* self, int index )
{
    void* ret = self->elements[index]; self->elements[index] = null;

    if ( (index == self->first) && (index == self->last) )
    {
        self->first = -1;
        self->last  = -1;
    }

    if ( index == self->first ) self->first++;

    if ( index == self->last  ) self->last--;

    if ( ret ) self->count--;

    return ret;
}
~

... Function: Array_resize

~source/c/Array.c~
static void Array_resize( Array* self, int index )
{
    int new_size = self->capacity;

    while ( new_size < (index + 1) )
    {
        new_size = new_size + new_size;
    }

    if ( self->capacity != new_size )
    {
        void** elements = NewArray( void*, new_size );

        for ( int i=0; i < self->capacity; i++ )
        {
            elements[i] = self->elements[i]; self->elements[i] = 0;
        }

        DeleteArray( &self->elements );

        self->elements = elements;
        self->capacity = new_size;
    }
}
~

~source/c/Array.c~
String*
Array_joinStrings_separator_number( const Array* self, char separator, int number )
{
    String*       joined = null;
    StringBuffer* buffer = StringBuffer_new();
    {
        char sep[2] = { separator, '\0' };
        int count = Array_getCount( self );

        if ( count )
        {
            for ( int i=0; i < count; i++ )
            {
                if ( number-- > 0 )
                {
                    StringBuffer_append_string( buffer, (String*) Array_get_index( self, i ) );

                    if ( number )
                    {
                        StringBuffer_append_chars( buffer, sep );
                    }
                }
            }
        }
    }
    joined = String_new( StringBuffer_getChars( buffer ) );

    Delete( &buffer );

    return joined;
}

bool
String_endsWith( const String* self, const char* suffix )
{
    const char* aString = self->data;

    int len1 = strlen( suffix );

    if ( 0 == len1 )
    {
        return 0;
    }
    else
    {
        int len2 = strlen( aString );

        if ( len1 > len2 )
        {
            return 0;
        }
        else
        {
            const char* _suffix = &(suffix [len1]);
            const char* _string = &(aString[len2]);

            do
            {
                _suffix--;
                _string--;

                if ( _suffix == suffix )
                {
                    return (*_suffix == *_string);
                }
            }
            while ( *_suffix == *_string );

            return 0;
        }
    }
}

bool
String_isURLPath( const String* self )
{
    const char* chars = String_getChars ( self );
    int         n     = String_getLength( self );
    char        ch;

    if ( '/' != chars[0] ) return false;

    for ( int i=0; i < n; i++ )
    {
        ch = chars[i];

        switch ( ch )
        {
        // eos
        case '\0':
            continue;

        // unreserved = ALPHA / DIGIT / "-" / "." / "_" / "~"
        case '-': //
        case '.': //
            if ( '.' == chars[i+1] ) return false;
        case '_': // 32
        case '~': //
            continue;

        // gen-delims = ":" / "/" / "?" / "#" / "[" / "]" / "@"
        case ':': //
        case '/': //
            if ( '/' == chars[i+1] ) return false;
        case '?': //
        case '#': //
        case '[': //
        case ']': //
        case '@': //
            continue;

        // sub-delims  = "!" / "$" / "&" / "'" / "(" / ")" / "*" / "+" / "," / ";" / "="
        case '!': //
        case '$': //
        case '&': //
        case '\'': //
        case '(': //
        case ')': //
        case '*': //
        case '+': //
        case ',': //
        case ';': //
        case '=': //
            continue;

        default:
            if ( ('a' <= ch) && (ch <= 'z') ) continue;
            if ( ('A' <= ch) && (ch <= 'Z') ) continue;
            if ( ('0' <= ch) && (ch <= '9') ) continue;
            return false;
        }
    }

    return true;
}

Array*
String_toArray_separator( const String* self, char separator )
{
    Array* parts = Array_new();

    int loop  = 1;
    int start = 0;
    int skip  = 0;

    do
    {
        int index = String_indexOf_ch_skip( self, separator, skip );

        if ( -1 != index )
        {
            int len = index - start;

            if ( len > 0 )
            {
                String* part = String_substring_index_length( self, start, len );
                Array_append_element( parts, (void**) &part );
            }

            start = index + 1;
            skip++;
        }
        else
        {
            String* part = String_substring_index( self, start );
            if ( String_getLength( part ) > 0 )
            {
                Array_append_element( parts, (void**) &part );
            }
            else
            {
                Delete( &part );
            }
            loop = 0;
        }
    }
    while ( loop );

    return parts;
}

String_HexEncode( const unsigned char *buffer, int len )
{
    char* encoded = Platform_Alloc( (len * 3) + 1, sizeof( char ) );
    char* tgt     = encoded;
    char  mask    = 0x0F;

    for ( int i=0; i < len; i++ )
    {

~source/c/StringBuffer.c~
void
StringBuffer_append_chars( StringBuffer* self, const char* chars )
{
    //  self->capacity = 16;
    //  self->length   = 8;
    //  self->data     = "Original00000000"
    //  chars          = "NewString"

    int charlen = CharString_length( chars );       // 9
    int newlen  = self->length + charlen;       // 17 = 8 + 9
    int oldcap  = self->capacity;           // 16

    while ( newlen > (self->capacity - 1) )
    {
        self->capacity *= 2;                // 32
    }

    if ( oldcap != self->capacity )
    {
                                        //  0123456789X123456789X123456789X12
        self->data = Platform_Realloc( self->data, self->capacity ); // "Original00000000################"

        for ( int i=oldcap; i < self->capacity; i++ )
        {
            self->data[i] = '\0';
        }
                                        //                  |<------------>|
                                        //  0123456789X123456789X123456789X12
                                        // "Original000000000000000000000000"
    }

    for ( int i=0; i < charlen; i++ )
    {
        self->data[self->length + i] = chars[i];
    }

                                        //          |<----->|
                                        //  0123456789X123456789X123456789X12
                                        // "OriginalNewString000000000000000"

    self->length = newlen;
}

static char* Join( const char* dirname, const char* basename )
{
    int  len_dirname  = strlen( dirname  );
    int  len_basename = strlen( basename );
    bool insert_slash = false;

    if
    (
        ('/' != dirname[len_dirname - 1])
        &&
        ('/' != basename[0] )
    )
    {
        len_dirname++;
        insert_slash = true;
    }

    int len = len_dirname + len_basename + 1;

    char* dst = NewArray( char, len );

    strncpy( dst, dirname, len_dirname );

    if ( insert_slash )
    {
        dst[len_dirname - 1] = '/';
    }

    strncpy( dst + len_dirname, basename, len_basename );

    return dst;
}

FGetLine( FILE* stream, size_t* len )
{
    String* line = NULL;
    {
        size_t sz     = 1024;
        char*  buffer = NewArray( char, sz );
        size_t i      = 0;
        bool   loop   = true;
        bool   eof    = false;

        do
        {
            if ( (sz - 1) == i )
            {
                sz *= 2;
                buffer = Platform_Realloc( buffer, sz * sizeof( char ) );
            }

            int ch = fgetc( stream );

            switch ( ch )
            {
            case EOF:
                loop = false;
                eof  = true;
                break;

            case '\n':
                loop        = false;
                if ( '\r' == buffer[i-1] )
                {
                    i--;
                    buffer[i] = '\0';
                }
                break;

            default:
                buffer[i++] = ch;
            }

        } while ( loop );

        if ( !eof )
        {
            line = String_new( buffer );

            if ( len ) *len = i;
        }

        DeleteArray( &buffer );
    }
    return line;
}

{
    int result = 0;
    do
    {
        const void* inner = Port_getInner( address );


        result = bind( self->descriptor, inner, sizeof( struct sockaddr ) );
        if (0 != result)
        {
            printf( "Port is busy, waiting(%i)...\n", wait );
            IO_printError( IO_StdErr() );
            Platform_SecondSleep( 1 );

bool
IO_accept( IO* self, Port* peer, IO** connection )
{
    socklen_t addrlen = sizeof( struct sockaddr );

    void* inner = Port_getInner( peer );

    int fd = accept( self->descriptor, inner, &addrlen );

    if ( -1 == fd )
    {
        switch ( errno )
        {
        case EBADF:
            //  Ignore as FD is invalidated when CTRL + C is pressed.
            break;

        default:
            fprintf( stdout, "IO_accept: error: %i - %s\n", errno, strerror( errno ) );
        }
        return false;
    }
    else
    {
        *connection = IO_open_mode( IO_new( &fd ), "r+" );

        return true;
    }
}

int
IO_receive( IO* self, char* buffer, int size, Port** port )
{
    struct    sockaddr    _remote_address;
    struct    sockaddr_in* remote_address = (struct sockaddr_in*) &_remote_address;

    socklen_t address_len = sizeof( struct sockaddr );
    int       flags       = MSG_WAITALL;
    int       count       = recvfrom( self->descriptor, buffer, size, flags, &_remote_address, &address_len );

    *port = Port_new( 0 );
    struct sockaddr_in* inner = (struct sockaddr_in*) Port_getInner( *port );

    inner->sin_family      = remote_address->sin_family;
    inner->sin_port        = remote_address->sin_port;
    inner->sin_addr.s_addr = remote_address->sin_addr.s_addr;

    if ( -1 == count )
    {
        switch ( errno )
        {
        case EBADF:
            //  Ignore as FD is invalidated when CTRL + C is pressed.
            break;

        default:
            fprintf( stdout, "IO_accept: error: %i - %s\n", errno, strerror( errno ) );
        }
        return 0;
    }
    else
    {
        return count;
    }
}

            if ( String_endsWith( extension, ".xhtml"  ) ) return "application/xhtml+xml";
            if ( String_endsWith( extension, ".xls"    ) ) return "application/vnd.ms-excel";
            if ( String_endsWith( extension, ".xlsx"   ) ) return "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet";
            if ( String_endsWith( extension, ".xml"    ) ) return "application/xml";
            if ( String_endsWith( extension, ".xul"    ) ) return "application/vnd.mozilla.xul+xml";
            break;

        case 'z':
            if ( String_endsWith( extension, ".zip"    ) ) return "application/zip";
            break;

        case '3':
            if ( String_endsWith( extension, ".3gp"    ) ) return "video/3gpp";
            if ( String_endsWith( extension, ".3g2"    ) ) return "video/3gpp2";
            break;

        case '7':
            if ( String_endsWith( extension, ".7z"     ) ) return "application/x-7z-compressed";
            break;

        default:
            break;
        }
    }
    return "plain/text";
}

~source/c/Security.c~
void
Security_DropPrivilegesOrAbort()
{
    uid_t  uid = getuid();
    uid_t euid = geteuid();
    gid_t  gid = getgid();
    gid_t egid = getegid();

    struct passwd* nobody = getpwnam( "nobody" );

    fprintf( stdout, "Existing privileges, uid: %i, euid: %i, gid: %i egid: %i\n", uid, euid, gid, egid );

    if ( 0 != uid )
    {
        fprintf( stdout, "Keeping existing privileges\n" );
    }
    else
    if ( !nobody )
    {
        fprintf( stdout, "Aborting: could not retrieve details of unprivileged 'nobody' account.\n" );
        abort();
    }
    else
    {
        gid_t groups[2];

        if ( 0 != setgid( nobody->pw_gid ) )
        {
            fprintf( stdout, "Aborting: error calling setgid to drop root group privileges.\n" );
            abort();
        }
        else
        if ( nobody->pw_gid != getgid() )
        {
            fprintf( stdout, "Aborting: group not changed to expected value, expected: %i, got: %i.\n", nobody->pw_gid, getgid() );
            abort();
        }
        else
        {
            fprintf( stdout, "Dropped root group privileges\n" );
        }

        groups[0] = nobody->pw_gid;

        if ( 0 != setgroups( 1, groups ) )
        {
            fprintf( stdout, "Aborting: error calling setgroups to drop root groups list.\n" );
            abort();
        }
        else
        if ( (1 != getgroups( 2, groups )) && (nobody->pw_gid == groups[0]) )
        {
            fprintf( stdout, "Aborting: group list contains more goups than the one expected, found %i groups.\n", getgroups( 0, groups ) );
            abort();
        }
        else
        {
            fprintf( stdout, "Dropped root supplementary groups privileges\n" );
        }

        if ( 0 != setuid( nobody->pw_uid ) )
        {
            fprintf( stdout, "Aborting: error calling setuid to drop root userprivileges.\n" );
            abort();
        }
        else
        if ( nobody->pw_uid != getgid() )
        {
            fprintf( stdout, "Aborting: user not changed to expected value, expected: %i, got: %i.\n", nobody->pw_uid, getuid() );
            abort();
        }
        else
        {
            fprintf( stdout, "Dropped root user privileges\n" );
        }
    }
}

bool
Service_listen_wait( Service* self, int wait )
{
    bool status = false;

    if ( IO_bind_address_wait( self->socket, self->portspec, wait ) )
    {
        printf( "Bound socket to any\n" );
        Security_DropPrivilegesOrAbort();

        if ( UDP == self->socketType )
        {
            return true;
        }
        else
        if ( TCP == self->socketType )
        {
            if ( IO_listen( self->socket ) )
            {
                printf( "Listening on port: %i\n", self->port );

                signal1  = signal(  1, SignalHandler );
                signal2  = signal(  2, SignalHandler );
                signal9  = signal(  9, SignalHandler );
                signal13 = signal( 13, IgnoreSigpipe );

                status = true;
            }
        }
    }
    return status;
}

void
Service_acceptConnections_fork( Service* self, bool fork )
{
    if ( !self->handler )
    {
        printf( "Aborting! No handler set using 'Service_setHandler_handler'." );
    }
    else
    {
        Port* peer       = Port_new( 0 );
        IO*   connection = NULL;
        Path* cwdDir     = Path_CurrentDirectory();

        while ( KeepAlive )
        {
            if ( IO_accept( self->socket, peer, &connection ) )
            {
                if ( fork )
                {
                    if ( Platform_Fork() )
                    {
                        Platform_Wait();
                    }
                    else
                    {
                       self->handler( cwdDir, peer, self->settings, connection, NULL, 0 );
                       KeepAlive = false;
                    }
                }
                else
                {
                    self->handler( cwdDir, peer, self->settings, connection, NULL, 0 );
                }
                Delete( &connection );
            }
        }

        Delete( &cwdDir );
        Delete( &peer   );
    }
}

Service_acceptPackets_fork( Service* self, bool fork )

    if ( !self->handler )

        printf( "Aborting! No handler set using 'Service_setHandler_handler'." );

        while ( KeepAlive )

                        Platform_Wait();
                    }
                    else
                    {
                        self->handler( cwdDir, peer, self->settings, self->socket, buffer, count );
                        KeepAlive = false;
                    }
                }
                else
                {
                    self->handler( cwdDir, peer, self->settings, self->socket, buffer, count );
                }
            }
        }

    }

static void validate( Line* self )
{
    const String* data  = Line_getData    ( self );
    const char*   chars = String_getChars ( data );
    int           n     = String_getLength( data );

    for ( int i=0; i < n; i++ )
    {
        unsigned int ch = chars[i];
    
        bool is_lowsym = (('!' <= ch) && (ch <= '@'));
        bool is_midsym = (('[' <= ch) && (ch <= '`'));
        bool is_uppsym = (('(' <= ch) && (ch <= '~'));
        bool is_digit  = (('0' <= ch) && (ch <= '9'));
        bool is_upper  = (('A' <= ch) && (ch <= 'Z'));
        bool is_lower  = (('a' <= ch) && (ch <= 'z'));
        bool is_alpha  = is_digit  || is_upper  || is_lower;
        bool is_symbol = is_lowsym || is_midsym || is_uppsym;

        if ( 127 < ch )
        {
            self->isASCII         = false;
            self->isNormal        = false;
            self->isNonWhitespace = false;
            self->isSymbolic      = false;
            self->isAlphanumeric  = false;
            self->isAlphabetic    = false;
            self->isNumeric       = false;
            self->isSafe          = false;
        }
        else
        if ( (ch < ' ') || (ch == 127) )
        {
            self->isNormal        = false;
            self->isNonWhitespace = false;
            self->isSymbolic      = false;
            self->isAlphanumeric  = false;
            self->isAlphabetic    = false;
            self->isNumeric       = false;
            self->isSafe          = false;
        }
        else
        if ( ch == ' ' )
        {
            self->isNonWhitespace = false;
            self->isSymbolic      = false;
            self->isAlphanumeric  = false;
            self->isAlphabetic    = false;
            self->isNumeric       = false;
        }
        else
        if ( !is_alpha )
        {
            self->isAlphanumeric  = false;
            self->isAlphabetic    = false;
            self->isNumeric       = false;
        }
        else
        if ( !is_upper && !is_lower )
        {
            self->isAlphabetic    = false;
        }
        else
        if ( !is_digit )
        {
            self->isNumeric       = false;
        }
        else
        if ( !is_symbol )
        {
            self->isSymbolic      = false;
}   }   }

~source/c/Tokenizer.c~
Token* Tokenizer_nextRealToken( Tokenizer* self, IO* io )
{
    Token* token = null;

    if ( self->readFully )
    {
        bool whitespace = true;

        do
        {
            token = Array_remove_index( self->tokens, self->next++ );

            if ( !token )
            {
                whitespace = false;
            }
            else
            if ( TOKEN_WHITESPACE == Token_getType( token ) )
            {
                if ( io )
                {
                    IO_print( io, String_getContent( Token_getContent( token ) ) );
                }

                Delete( &token );
            }
            else
            {
                whitespace = false;
            }
        }
        while ( whitespace );
    }

    return token;
}

Array*
Tokenizer_fracture( Tokenizer* self, IO* stream, const char** startNames )
{
    Array* parts   = Array_new();
    Array* current = Array_new();
    Token* tmp     = null;

    while ( (tmp = Tokenizer_parseToken( self, stream )) )
    {
        if ( CharStringArray_Contains( startNames, String_getContent( Token_getContent( tmp ) ) ) )
        {
            Array_append_element( parts, Give( current ) );

            current = Array_new();
        }

        Array_append_element( current, Give( tmp ) );
    }

    Array_append_element( parts, Give( current ) );

    return parts;
}

~source/c/Tokenizer.c~
bool Tokenizer_isKeyword( const Tokenizer* self, const Token* token )
{
    bool         is_keyword = false;
    const char** tmp        = null;
    const char*  keyword    = null;

    if ( token && self->keywords )
    {
        tmp = self->keywords;

        while ( (keyword = *tmp++) )
        {
            if ( String_contentEquals( token->content, keyword ) )
            {
                is_keyword = true;
                break;
            }
        }
    }
    return is_keyword;
}

~source/c/Tokenizer.c~
Token* Tokenizer_parseToken( Tokenizer* self, IO* stream )
{
    Token* token = Tokenizer_ParseToken( stream );

    if ( token )
    {
        token->line   = self->line;
        token->column = self->column;

        if ( TOKEN_WHITESPACE == Token_getType( token ) )
        {
            const char* whitespace = String_getContent( Token_getContent( token ) );

            while ( *whitespace )
            {
                switch ( *whitespace )
                {
                case '\n':
                    self->line++;
                    self->column = 1;
                    break;

                default:
                    self->column++;
                }
                whitespace++;
            }
        }
        else
        {
            self->column += String_getLength( Token_getContent( token ) );
        }

        if ( Tokenizer_isKeyword( self, token ) )
        {
            token->type = TOKEN_KEYWORD;
        }
    }
    return token;
}

~source/c/Tokenizer.c~
Token* _ParseSymbol( IO* stream, char first )
{
    Token*        token = null;
    StringBuffer* sb    = StringBuffer_new();
    int           type  = 0;
    bool          loop  = true;
    char          ch;

    StringBuffer_append_char( sb, first );
    
    while ( loop && Character_isSymbol( (ch = IO_readChar( stream )) ) )
    {
        StringBuffer_append_char( sb, ch );

        switch ( ch )
        {
        case '{':
        case '(':
        case '[':
            loop = false;
            break;

        case '}':
        case ')':
        case ']':
            loop = false;
            break;

        default:
            break;
        }
    }

    IO_unread( stream, ch );

    switch ( first )
    {
    case '{':
    case '(':
    case '[':
        type = TOKEN_OPEN;
        loop = false;
        break;

    case '}':
    case ')':
    case ']':
        type = TOKEN_CLOSE;
        loop = false;
        break;

    case '!':
    case '@':
    case '#':
    case '$':
    case '^':
    case '&':
    case '\\':
        type = TOKEN_SYMBOL_PREFIX;
        break;

    case '=':
    case '|':
    case ':':
    case '.':
    case '/':
        type = TOKEN_SYMBOL_INFIX;
        break;

    case '-':
    case '+':
        type = TOKEN_SYMBOL_PREORPOSTFIX;
        break;

    case '%':
    case '*':
    case '?':
        type = TOKEN_SYMBOL_CONTEXTUAL;
        break;

    default:
        type = TOKEN_SYMBOL;
    }

    token       = Token_new( (String**) Platform_Give( String_new( StringBuffer_getChars( sb ) ) ) );
    token->type = type;

    Delete( &sb );

    return token;
}

Token* _ParseWhitespace( IO* stream, char first )
{
    Token*        token = null;
    StringBuffer* sb    = StringBuffer_new();
    int           type  = TOKEN_WHITESPACE;
    bool          loop  = true;
    char          ch;

    StringBuffer_append_char( sb, first );

    if ( Character_isNewline( first ) )
    {
        type = TOKEN_NEWLINE;
    }
    else
    {
        while ( loop && Character_isWhitespace( (ch = IO_readChar( stream )) ) )
        {
            if ( !Character_isNewline( ch ) )
            {
                StringBuffer_append_char( sb, ch );
            }
            else
            {
                loop = false;
            }
        }
        IO_unread( stream, ch );
    }

    token       = Token_new( (String**) Platform_Give( String_new( StringBuffer_getChars( sb ) ) ) );
    token->type = type;

    Delete( &sb );

    return token;
}

{
    const Token* peek  = null;
          Token* token = null;

    while ( loop && (peek = Tokenizer_peekToken( tokenizer )) )
    {

~source/c/Table.c~
void Table_replace_valist( Table* self, ... )
{
    int     d;
    char*   s;

    int     n = Row_getFieldCount( self->spec );
    va_list values;
    Tuple*  t = New( Tuple );
            t->count = n;
            t->ftypes = NewArray( FieldType, n );
            t->values = NewArray( void*,     n );

    va_start( values, self );
 
    for ( int i=0; i < n; i++ )
    {
        FieldType ft = Row_getFieldType_at( self->spec, i );

        switch ( ft )
        {
        case INT:
            d = va_arg( values, int );
            //Console_Print_pattern_valist( "%i\n", d );
            t->ftypes[i] = ft;
            t->values[i] = INT2VOIDP( d );
            break;
        
        case TEXT:
            s = va_arg( values, char * );
            //Console_Print_pattern_valist( "%s\n", s );
            t->ftypes[i] = ft;
            t->values[i] = CharString_copy( s );
            break;

        case UNKNOWN:
            va_arg( values, char * );
            t->ftypes[i] = ft;
            t->values[i] = 0;
            break;
        }
    }
    va_end( values );

    Array_append_element( self->rows, (void**) &t );
}
~

~source/c/Table.c~
void Table_replace_keyvals( Table* self, ... )
{
    int     n = Row_getFieldCount( self->spec );
    Tuple*  t = New( Tuple );
            t->count = n;
            t->ftypes = NewArray( FieldType, n );
            t->values = NewArray( void*,     n );

    va_list   keyvals;
    va_start( keyvals, self );
    
    KeyVal keyval = va_arg( keyvals, KeyVal );
    
    while ( keyval.key )
    {
        int       at = Row_getFieldPositionFor_name( self->spec, keyval.key );
        FieldType ft = Row_getFieldType_at         ( self->spec, at         );

        t->ftypes[at] = ft;

        switch ( ft )
        {
        case INT:
            t->values[at] = keyval.val;
            Console_Print_pattern_valist( "%s=%d\n", keyval.key, keyval.val );
            break;
        
        case TEXT:
            t->values[at] = CharString_copy( keyval.val );
            Console_Print_pattern_valist( "%s=%s\n", keyval.key, keyval.val );
            break;
        
        case UNKNOWN:
        default:
            t->values[at] = 0;
            Console_Print_pattern_valist( "%s=?\n", keyval.key );
            break;
        }

        keyval = va_arg( keyvals, KeyVal );
    }
    
    va_end( keyvals );
    
    Array_append_element( self->rows, (void**) &t );
}

~source/c/Table.c~
void Table_empty( Table* self )
{
    int n = Array_getCount( self->rows );
    
    for ( int i=0; i < n; i++ )
    {
        Tuple* t = Array_remove_index( self->rows, i );

        if ( t )
        {
            int m = Row_getFieldCount( self->spec );
            
            for ( int j=0; j < m; j++ )
            {
                FieldType ft = Row_getFieldType_at( self->spec, j );

                switch( ft )
                {
                case TEXT:
                    DeleteArray( &(t->values[j]) );
                    break;

                default:
                    break;
                }
            }

            DeleteArray( &t->ftypes );
            DeleteArray( &t->values );
            Delete     ( &t         );
        }
    }
}

~source/c/Table.c~
Array*
Table_select_valist( const Table* self, ... )
{
    Array* results = Array_new();
    {
        va_list where;
        bool    loop = true;
        char*   ch;

        va_start( where, self );
        
        while ( loop )
        {
            ch = va_arg( where, char* );
            
            if ( ch )
            {
                Console_Print_pattern_valist( "%s\n", ch );
            }
            else
            {
                loop = false;
            }
        }

        va_end( where );
    }

    int n = Array_getCount( self->rows );

    for ( int i=0; i < n; i++ )
    {
        Row* copy = Row_copy( self->spec );
        {
            const Tuple* t = Array_get_index( self->rows, i );
            int   m        = t->count;
            
            for ( int j=0; j < m; j++ )
            {
                String* str_value = null;
                
                FieldType ft = t->ftypes[j];

                switch( ft )
                {
                case TEXT:
                    str_value = String_new( t->values[j] );
                    break;
                
                case INT:
                    str_value = String_new_value( (long) t->values[j] );
                    break;
                    
                case UNKNOWN:
                default:
                    str_value = String_new( "" );
                    break;
                }
                Row_appendValue_string( copy, &str_value );
            }
        }
        Array_append_element( results, (void**) &copy );
    }
    
    return results;
}

void
HTTPRequest_log_status( const HTTPRequest* self, const char* status, int statusNum )
{
    //
    //  [xff 192.168.1.1:12345] [client 192.168.1.162:53758] Duration: ms. GET host:/url/ status
    //61

    char   origin[1024];
    char      xff[1024];
    char   method[1024];
    char     host[1024];
    char resource[1024];
    char  logline[6144];

    snprintf(   origin, 1023, "[origin %s]", String_getChars( HTTPRequest_getOrigin      ( self ) ) );
    snprintf(      xff, 1023, "[xff %s]",    String_getChars( HTTPRequest_getForwardedFor( self ) ) );
    snprintf(   method, 1023, "%5s",         String_getChars( HTTPRequest_getMethod      ( self ) ) );
    snprintf(     host, 1023, "%s",          String_getChars( HTTPRequest_getHost        ( self ) ) );
    snprintf( resource, 1023, "%s",          String_getChars( HTTPRequest_getResource    ( self ) ) );

    if ( 0 < String_getLength( HTTPRequest_getForwardedFor( self ) ) )
    {
        snprintf( logline, 6143, "[%3i] %s   %s %s %s%s", statusNum, origin, xff, method, host, resource );
    }
    else
    {
        snprintf( logline, 6143, "[%3i] %-24s %s %s%s", statusNum, origin, method, host, resource );
    }

    SysLog_Log_chars( logline );
}

~source/c/HTTP.c~
HTTPRequest*
HTTPRequest_Parse( const Port* peer, IO* connection, const String* localDomain )
{
    bool read_body = true;

    HTTPRequest* request      = HTTPRequest_new();
    String*      request_line = HTTPRequest_ReadLine( connection );

    if ( request_line )
    {
        HTTPRequestLine* http_request_line = HTTPRequestLine_new( request_line );

        if ( HTTPRequestLine_isValid( http_request_line ) )
        {
            String* origin = Port_origin( peer );
            {
                HTTPRequest_setStartLine( request, String_getChars( request_line                                     ) );
                HTTPRequest_setMethod   ( request, String_getChars( HTTPRequestLine_getMethod  ( http_request_line ) ) );
                HTTPRequest_setResource ( request, String_getChars( HTTPRequestLine_getResource( http_request_line ) ) );
                HTTPRequest_setVersion  ( request, String_getChars( HTTPRequestLine_getVersion ( http_request_line ) ) );
                HTTPRequest_setOrigin   ( request, String_getChars( origin                                           ) );
            }
            Delete( &origin );

            String* line;
            bool loop = true;
            do
            {
                if ( (line = HTTPRequest_ReadLine( connection )) )
                {
                    String_trimEnd( line );
                    {
                        int len = String_getLength( line );
                        {
                            HTTPHeader* header = HTTPHeader_new( &line );

                            if ( String_contentEquals( header->name, "Host" ) )
                            {
                                const String* value = HTTPHeader_getValue( header );

                                if ( String_contains( value, ":" ) )
                                {
                                    int     index = String_indexOf_ch_skip       ( value,   ':',     0 );
                                    String* host  = String_substring_index_length( value,     0, index );
                                    String* port  = String_substring_index       ( value, index        );
                                    {
                                        HTTPRequest_setHost( request, String_getChars( host ) );
                                        HTTPRequest_setPort( request, String_getChars( port ) );
                                        {
                                            int     dot_index  = String_indexOf_ch_skip( host, '.', 0 );
                                            String* first_quad = String_substring_index_length( host, 0, dot_index );

                                            if ( String_isNumeric( first_quad ) )
                                            {
                                                int number = String_toNumber_default( first_quad, 255 );

                                                if ( number < 255 ) request->ip_target = true;

                                                if ( localDomain )
                                                {
                                                    HTTPRequest_setHost( request, String_getChars( localDomain ) );
                                                    request->ip_target = false;
                                                }
                                            }
                                            Delete( &first_quad );
                                        }
                                    }
                                    Delete( &host );
                                    Delete( &port );
                                }
                                else
                                {
                                    HTTPRequest_setHost( request, String_getChars( value ) );
                                }
                            }
                            else
                            if ( String_contentEquals( header->name, "Origin" ) )
                            {
                                const String* value = HTTPHeader_getValue( header );

                                HTTPRequest_setOrigin( request, String_getChars( value ) );
                            }
                            else
                            if ( String_contentEquals( header->name, "X-Forwarded-For" ) )
                            {
                                const String* value = HTTPHeader_getValue( header );

                                HTTPRequest_setForwardedFor( request, String_getChars( value ) );
                            }
                            else
                            if ( String_contentEquals( header->name, "Authorization" ) )
                            {
                                const String* value = HTTPHeader_getValue( header );

                                HTTPRequest_setAuthorization( request, String_getChars( value ) );
                            }
                            else
                            if ( String_contentEquals( header->name, "Content-Length" ) )
                            {
                                const String* value          = HTTPHeader_getValue( header );
                                long          content_length = strtol( String_getChars( value ), NULL, 10 );

                                HTTPRequest_setContentLength( request, content_length );
                            }

                            //printf( stdout, "[%s][%s]\n", String_getChars( header->name ), String_getChars( header->value ) );

                            Array_append_element( request->headers, (void**) &header );
                        }
                        if ( 0 == len ) loop = false;
                    }
                }
                else
                {
                    loop = false;
                }
            } while ( loop );
        }

        long content_length = HTTPRequest_getContentLength( request );
        if ( read_body && content_length )
        {
            String* body = IO_readdata( connection, sizeof(char), content_length );

            HTTPRequest_setBody( request, String_getChars( body ) );

            Delete( &body );
        }
        Delete( &http_request_line );

        //Delete( &test );
        //Delete( &rl   );

        HTTPRequest_validate( request );
    }
    Delete( &request_line );

    return request;
}

String*
HTTPHeader_ExtractHeaderValue( const String* line )
{
    //  012345678901234567890
    //  Host: 127.0.0.1:8080

    if ( String_getLength( line ) )
    {
        const char* chx   = String_getChars       ( line         );
              int   len   = String_getLength      ( line         );
              int   index = String_indexOf_ch_skip( line, ':', 0 ) + 1;

        while ( (index < len) && (' ' == chx[index]) ) index++;

        return (index < len) ? String_substring_index( line, index ) : String_new( "" );
    }
    else
    {
        return String_new( "" );
    }
}

~source/c/HTTP.c~
HTTPRequestLine*
HTTPRequestLine_new( String* line )
{
    HTTPRequestLine* self = New( HTTPRequestLine );
    if ( self )
    {
        Object_init( &self->super, (Destructor) HTTPRequestLine_destruct );

        int end_method   = String_indexOf_ch_skip( line, ' ', 0 );
        int end_resource = String_indexOf_ch_skip( line, ' ', 1 );
        int len          = end_resource - (end_method + 1);

        if ( end_method >= 3 )
        {
            String* method = String_substring_index_length( line, 0, end_method );
            if
            (
                String_contentEquals( method, "POST"    )
                ||
                String_contentEquals( method, "GET"     )
                ||
                String_contentEquals( method, "HEAD"    )
                ||
                String_contentEquals( method, "OPTIONS" )
            )
            {
                self->method = Take( &method );
            }
            Delete( &method );
        }

        if ( len >= 1 )
        {
            String* resource = String_substring_index_length( line, end_method + 1, len );
            if ( String_isURLPath( resource ) )
            {
                self->resource = Take( &resource );
            }
            Delete( &resource );
        }

        if ( len >= 1 )
        {
            String* version = String_substring_index( line, end_resource + 1 );
            if
            (
                String_contentEquals( version,  "HTTP/1.0" )
                ||
                String_contentEquals( version,  "HTTP/1.1" )
            )
            {
                self->version = Take( &version );
            }
            Delete( &version );
        }
    }
    return self;
}

~source/c/HTTP.c~
void HTTPBasicCredentials_init( HTTPBasicCredentials* self, const String* authorization_header )
{
    self->username = String_new( "" );
    self->password = String_new( "" );
    self->isValid  = false;

    if ( authorization_header )
    {
        Array* auth_bits = String_toArray_separator( authorization_header, ' ' );
        {
            if ( 2 == Array_getCount( auth_bits ) )
            {
                const String* type = Array_get_index( auth_bits, 0);

                if ( String_contentEquals( type, "Basic" ) )
                {
                    const String* encoded = Array_get_index( auth_bits, 1 );

                    if ( encoded )
                    {
                        const String* credentials = credentials = Base64Decode( encoded );

                        if ( credentials )
                        {
                            Array* username_and_password = String_toArray_separator( credentials, ':' );

                            if ( 2 == Array_getCount( username_and_password ) )
                            {
                                self->username = String_copy( Array_get_index( username_and_password, 0 ) );
                                self->password = String_copy( Array_get_index( username_and_password, 1 ) );
                                self->isValid  = true;
                            }

                            Delete( &username_and_password );
                        }
                    }
                }
            }
        }
        Delete( &auth_bits );
    }
}

int Base64decode_len(const char *bufcoded)
{
    int nbytesdecoded;
    register const unsigned char *bufin;
    register int nprbytes;

    bufin = (const unsigned char *) bufcoded;
    while (pr2six[*(bufin++)] <= 63);

    nprbytes = (bufin - (const unsigned char *) bufcoded) - 1;
    nbytesdecoded = ((nprbytes + 3) / 4) * 3;

    return nbytesdecoded + 1;
}

int Base64decode(char *bufplain, const char *bufcoded)
{
    int nbytesdecoded;
    register const unsigned char *bufin;
    register unsigned char *bufout;
    register int nprbytes;

    bufin = (const unsigned char *) bufcoded;
    while (pr2six[*(bufin++)] <= 63);
    nprbytes = (bufin - (const unsigned char *) bufcoded) - 1;
    nbytesdecoded = ((nprbytes + 3) / 4) * 3;

    bufout = (unsigned char *) bufplain;
    bufin = (const unsigned char *) bufcoded;

    while (nprbytes > 4) {
    *(bufout++) =
        (unsigned char) (pr2six[*bufin] << 2 | pr2six[bufin[1]] >> 4);
    *(bufout++) =
        (unsigned char) (pr2six[bufin[1]] << 4 | pr2six[bufin[2]] >> 2);
    *(bufout++) =
        (unsigned char) (pr2six[bufin[2]] << 6 | pr2six[bufin[3]]);
    bufin += 4;
    nprbytes -= 4;
    }

    /* Note: (nprbytes == 1) would be an error, so just ingore that case */
    if (nprbytes > 1) {
    *(bufout++) =
        (unsigned char) (pr2six[*bufin] << 2 | pr2six[bufin[1]] >> 4);
    }
    if (nprbytes > 2) {
    *(bufout++) =
        (unsigned char) (pr2six[bufin[1]] << 4 | pr2six[bufin[2]] >> 2);
    }
    if (nprbytes > 3) {
    *(bufout++) =
        (unsigned char) (pr2six[bufin[2]] << 6 | pr2six[bufin[3]]);
    }

    *(bufout++) = '\0';
    nbytesdecoded -= (4 - nprbytes) & 3;
    return nbytesdecoded;
}

static const char basis_64[] =
    "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

int Base64encode_len(int len)
{
    return ((len + 2) / 3 * 4) + 1;
}

int Base64encode(char *encoded, const char *string, int len)
{
    int i;
    char *p;

    p = encoded;
    for (i = 0; i < len - 2; i += 3) {
    *p++ = basis_64[(string[i] >> 2) & 0x3F];
    *p++ = basis_64[((string[i] & 0x3) << 4) |
                    ((int) (string[i + 1] & 0xF0) >> 4)];
    *p++ = basis_64[((string[i + 1] & 0xF) << 2) |
                    ((int) (string[i + 2] & 0xC0) >> 6)];
    *p++ = basis_64[string[i + 2] & 0x3F];
    }
    if (i < len) {
    *p++ = basis_64[(string[i] >> 2) & 0x3F];
    if (i == (len - 1)) {
        *p++ = basis_64[((string[i] & 0x3) << 4)];
        *p++ = '=';
    }
    else {
        *p++ = basis_64[((string[i] & 0x3) << 4) |
                        ((int) (string[i + 1] & 0xF0) >> 4)];
        *p++ = basis_64[((string[i + 1] & 0xF) << 2)];
    }
    *p++ = '=';
    }

    *p++ = '\0';
    return p - encoded;
}
